cmake_minimum_required(VERSION 3.24)
project(CommunityGraphicsEngine VERSION 0.3.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "CGE requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(IMGUI "${CMAKE_CURRENT_SOURCE_DIR}/third_party/imgui")
add_library(cge_imgui STATIC "${IMGUI}/imgui.cpp" "${IMGUI}/imgui_draw.cpp"
  "${IMGUI}/imgui_tables.cpp" "${IMGUI}/imgui_widgets.cpp" "${IMGUI}/backends/imgui_impl_dx11.cpp")
target_include_directories(cge_imgui PUBLIC "${IMGUI}" "${IMGUI}/backends")
target_compile_definitions(cge_imgui PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/shaders/CGE.hlsl" CGE_SHADER)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/shaders/CGE.hlsl")
configure_file(src/shader_source.h.in "${CMAKE_CURRENT_BINARY_DIR}/generated/shader_source.h" @ONLY)
add_library(cge_core STATIC src/swapchain.cpp src/graphics_host.cpp src/renderer.cpp
  src/settings.cpp src/editor.cpp src/editor_view.cpp src/input_capture.cpp)
target_include_directories(cge_core PUBLIC src PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(cge_core PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(cge_core PRIVATE /W4 /WX /permissive- /utf-8)
target_link_libraries(cge_core PUBLIC cge_imgui d3dcompiler dxguid user32)
add_library(cge SHARED src/skse_plugin.cpp src/startup_hook.cpp src/version.rc)
set_target_properties(cge PROPERTIES OUTPUT_NAME CGE)
target_compile_options(cge PRIVATE /W4 /WX /permissive- /utf-8)
target_link_libraries(cge PRIVATE cge_core)
target_link_options(cge PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/CGE.map")
include(CTest)
if(BUILD_TESTING)
  # The former relay is an isolated regression fixture, never installed or packaged.
  add_library(cge_test_proxy SHARED src/proxy.cpp src/d3d11.def)
  set_target_properties(cge_test_proxy PROPERTIES OUTPUT_NAME CGE-test-proxy)
  target_link_libraries(cge_test_proxy PRIVATE cge_core)
  add_executable(cge_smoke tests/smoke.cpp)
  target_compile_definitions(cge_smoke PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(cge_smoke PRIVATE d3dcompiler user32)
  add_test(NAME cge_d3d11_relay COMMAND cge_smoke $<TARGET_FILE:cge_test_proxy>)
  add_executable(cge_input_test tests/input_capture_test.cpp src/input_capture.cpp)
  target_include_directories(cge_input_test PRIVATE src)
  target_compile_definitions(cge_input_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(cge_input_test PRIVATE dxguid user32)
  add_test(NAME cge_independent_input COMMAND cge_input_test)
  add_executable(cge_keys_test tests/editor_keys_test.cpp)
  target_include_directories(cge_keys_test PRIVATE src)
  add_test(NAME cge_shortcuts COMMAND cge_keys_test)
endif()
install(TARGETS cge RUNTIME DESTINATION SKSE/Plugins)
install(FILES CGE.ini DESTINATION SKSE/Plugins)
install(FILES README.md LICENSE THIRD-PARTY-NOTICES.txt DESTINATION .)
install(FILES "${IMGUI}/LICENSE.txt" DESTINATION licenses RENAME DearImGui-LICENSE.txt)

install(FILES licenses/stb-MIT.txt licenses/ProggyFonts-MIT.txt DESTINATION licenses)
if(BUILD_TESTING)
  add_executable(cge_native_test tests/native_render_test.cpp)
  target_compile_definitions(cge_native_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(cge_native_test PRIVATE user32)
  add_test(NAME cge_native_render_unicode COMMAND cge_native_test $<TARGET_FILE:cge_test_proxy> "${CMAKE_CURRENT_BINARY_DIR}/native-unicode" --unicode)
  add_test(NAME cge_native_render_ansi COMMAND cge_native_test $<TARGET_FILE:cge_test_proxy> "${CMAKE_CURRENT_BINARY_DIR}/native-ansi" --ansi)
endif()
if(BUILD_TESTING)
  add_executable(cge_settings_test tests/settings_test.cpp src/settings.cpp)
  target_include_directories(cge_settings_test PRIVATE src)
  target_compile_definitions(cge_settings_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(cge_settings_test PRIVATE /utf-8)
  add_test(NAME cge_settings COMMAND cge_settings_test "${CMAKE_CURRENT_BINARY_DIR}/settings-fixture")
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME cge_distribution_boundary COMMAND "${Python3_EXECUTABLE}"

if(BUILD_TESTING)
  add_test(NAME cge_skse_rejects_root_injector COMMAND cge_skse_reject_test $<TARGET_FILE:cge>
